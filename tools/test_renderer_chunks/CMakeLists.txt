cmake_minimum_required(VERSION 3.17)
project(novacube_chunk_checks LANGUAGES C)
# (AI-assisted) CPU-only checks exercise the renderer with fake Vulkan/VMA endpoints, never a game window.
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
foreach(NC_CONFIG RELEASE RELWITHDEBINFO MINSIZEREL)
    string(REGEX REPLACE "[-/]DNDEBUG" "" CMAKE_C_FLAGS_${NC_CONFIG} "${CMAKE_C_FLAGS_${NC_CONFIG}}")
endforeach()
find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3-shared)
find_package(Vulkan REQUIRED)
get_filename_component(NC_ROOT "../.." ABSOLUTE)
add_executable(chunk_checks test.c "${NC_ROOT}/src/mesher.c")
target_include_directories(chunk_checks PRIVATE
        "${NC_ROOT}/include" "${NC_ROOT}/vendor/cvkm" "${NC_ROOT}/vendor/tds/include"
        "${NC_ROOT}/vendor/vma" "${NC_ROOT}/vendor/volk" "${NC_ROOT}/vendor/rapidhash" ${Vulkan_INCLUDE_DIRS})
target_link_libraries(chunk_checks PRIVATE SDL3::SDL3)
if(MSVC)
    target_compile_definitions(chunk_checks PRIVATE _CRT_SECURE_NO_WARNINGS)
    set_property(TARGET chunk_checks PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    target_compile_options(chunk_checks PRIVATE /W4 /WX /Zc:preprocessor /Gy /Gw /arch:AVX2)
    target_link_options(chunk_checks PRIVATE /OPT:REF)
else()
    target_compile_options(chunk_checks PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
    target_link_options(chunk_checks PRIVATE -Wl,--gc-sections)
    target_link_libraries(chunk_checks PRIVATE m)
endif()
if(WIN32)
    add_custom_command(TARGET chunk_checks POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:chunk_checks>)
endif()
enable_testing()
add_test(NAME chunk_checks COMMAND chunk_checks)
